#NAME User05.txt 
#LABEL "Center-finder (outside) with 3D-probe"
(Sucht den Mittelpunkt eines Außenrechtecks oder Kreises)
(Der 3D-Taster muss sich außerhalb des Werkstücks befinden)
(Es werden keine Nullpunkte gesetzt, sondern das Zentrum nur angefahren)
(Berücksichtigt den Tastkugel-Offset aus #700 und #701)
% 
#1=#954 (Anfahrgeschwindigkeit)
#2=#956  (max. Abmessung des Werkstücks)
#3=#958  (Toleranz)
IF #1=0 | #2=0 | #3=0 THEN
  PRINT "undefined macro parameters, please check variables #954, #956, #958"
  M2
ENDIF

(Prüfung ob Tastkugel-Offset-Parameter gesetzt sind)
IF #700=0 & #701=0 THEN
  PRINT "Probe ball offset not calibrated! Please run calibration macro first."
  PRINT "Variables #700 (X-offset) and #701 (Y-offset) must be set."
  M2
ENDIF

G21 G40 G90 G94
ASKMULTI "Outside Center finder";"User05.png";"Start point";"3D-probe must be located outside the workpiece" I=2
ASKVAL "Maximum workpiece dimension" P0=#2
ASKVAL "Safety distance for positioning" P1=10
#2=P0
#13=P1

PRINT "Starting outside center finder..."
PRINT "Maximum workpiece dimension: ";#2
PRINT "Safety distance: ";#13
PRINT "Using probe ball offset: X=";#700;" Y=";#701

(=== Erste Y-Richtung Messung ===)
#0=Y (Startpunkt Y)
#12=X (Startpunkt X)

(Y+ Kante antasten von unten)
G1 F=#1 UNTIL #I9=1 Y=Y+#2
IF #I9=0 THEN 
  PRINT "Y+ edge not found"
  M2 
ENDIF
G1 F=#1/10 UNTIL #I9=0 Y=Y-#3
IF #I9=1 THEN 
  PRINT "probe switch not released"
  M2 
ENDIF
#4=Y-#955/2-#701 (Y+ Kante korrigiert mit Tastkugel und Offset)

(Position für Y- Antastung: von Y+ Kante + Werkstück + Safety)
G0 Y=#4+#2+#13 (Y+ Kante + max. Abmessung + Safety)

(Y- Kante antasten von oben)
G1 F=#1 UNTIL #I9=1 Y=#4-#2
IF #I9=0 THEN 
  PRINT "Y- edge not found"
  M2 
ENDIF
G1 F=#1/10 UNTIL #I9=0 Y=Y+#3
IF #I9=1 THEN 
  PRINT "probe switch not released"
  M2 
ENDIF
#5=Y+#955/2+#701 (Y- Kante korrigiert mit Tastkugel und Offset)

(Y-Mittelpunkt anfahren)
G0 Y=(#4+#5)/2
PRINT "Y-center found between ";#5;" and ";#4;" = ";(#4+#5)/2

(=== X-Richtung Messung ===)
#0=X (Startpunkt X)

(X+ Kante antasten von links)
G1 F=#1 UNTIL #I9=1 X=X+#2
IF #I9=0 THEN 
  PRINT "X+ edge not found"
  M2 
ENDIF
G1 F=#1/10 UNTIL #I9=0 X=X-#3
IF #I9=1 THEN 
  PRINT "probe switch not released"
  M2 
ENDIF
#6=X-#955/2-#700 (X+ Kante korrigiert mit Tastkugel und Offset)

(Position für X- Antastung: von X+ Kante + Werkstück + Safety)
G0 X=#6+#2+#13 (X+ Kante + max. Abmessung + Safety)

(X- Kante antasten von rechts)
G1 F=#1 UNTIL #I9=1 X=#6-#2
IF #I9=0 THEN 
  PRINT "X- edge not found"
  M2 
ENDIF
G1 F=#1/10 UNTIL #I9=0 X=X+#3
IF #I9=1 THEN 
  PRINT "probe switch not released"
  M2 
ENDIF
#7=X+#955/2+#700 (X- Kante korrigiert mit Tastkugel und Offset)

(X-Mittelpunkt anfahren)
G0 X=(#6+#7)/2
PRINT "X-center found between ";#7;" and ";#6;" = ";(#6+#7)/2

(=== Zweite Y-Richtung Messung für höhere Genauigkeit ===)
(Zum Y-Mittelpunkt fahren für finale Messung)
G0 Y=(#4+#5)/2

(Y+ Kante antasten von unten)
G1 F=#1 UNTIL #I9=1 Y=Y+ABS(#4-(#4+#5)/2)+#13
IF #I9=0 THEN 
  PRINT "Y+ edge not found (second measurement)"
  M2 
ENDIF
G1 F=#1/10 UNTIL #I9=0 Y=Y-#3
IF #I9=1 THEN 
  PRINT "probe switch not released"
  M2 
ENDIF
#4=Y-#955/2-#701 (Y+ Kante korrigiert)

(Position für Y- Antastung)
G0 Y=#4+ABS(#4-#5)+#13

(Y- Kante antasten von oben)  
G1 F=#1 UNTIL #I9=1 Y=#4-ABS(#4-#5)-#13
IF #I9=0 THEN 
  PRINT "Y- edge not found (second measurement)"
  M2 
ENDIF
G1 F=#1/10 UNTIL #I9=0 Y=Y+#3
IF #I9=1 THEN 
  PRINT "probe switch not released"
  M2 
ENDIF
#5=Y+#955/2+#701 (Y- Kante korrigiert)

(Finaler Mittelpunkt)
G0 Y=(#4+#5)/2
G0 X=(#6+#7)/2

PRINT "=== CENTER FOUND ==="
PRINT "Final center position: X=";X;" Y=";Y
PRINT "Workpiece dimensions: X=";ABS(#6-#7);" Y=";ABS(#4-#5)
PRINT "Outside center finder completed successfully"

RETURN 

Simulation: 
PRINT "Outside Center-finder with 3D-probe (simulation)"
PRINT "This macro would find the center of an outside rectangle/circle"
RETURN
